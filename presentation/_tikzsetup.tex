\tikzset{
  medtube/.style args={#1/#2}{ % #1=color, #2=height
    draw, fill=#1,
    shape=rectangle,
    rounded corners=1mm,
    minimum width=0.8cm,
    minimum height=#2,
    path picture={
      % draw neck & cap inside the rectangle shape
      \draw[fill=#1!70!black] 
        ([xshift=0.1cm,yshift=-0.3cm]path picture bounding box.north west) -- 
        ([xshift=0.7cm,yshift=-0.3cm]path picture bounding box.north west) -- 
        ([xshift=0.6cm,yshift=0]path picture bounding box.north west) -- 
        ([xshift=0.2cm,yshift=0]path picture bounding box.north west) -- cycle;
      % inner "liquid" fill with lighter color
      \fill[#1!50] 
        ([xshift=0.2cm,yshift=-#2*0.7]path picture bounding box.north west) rectangle
        ([xshift=0.6cm,yshift=0]path picture bounding box.north west);
      % small label line
      \draw[white, line width=0.5pt]
        ([xshift=0.2cm,yshift=-#2*0.3]path picture bounding box.north west) -- 
        ([xshift=0.6cm,yshift=-#2*0.3]path picture bounding box.north west);
    }
  }
}

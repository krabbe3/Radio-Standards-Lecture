\begin{frame}{Electronic Measurements and Signal Processing Group \ac{acro:ems}}
  \textbf{We investigate the behavior of radio propagation channels\\ and develop methods to make them usable.}
  \vspace{0.5cm}
  \begin{columns}
    \begin{column}{0.35\textwidth}
      \hfill\includegraphics[width=0.9\textwidth]{images/section-0/Channel_sounder.png}
    \end{column}
    \begin{column}{0.65\textwidth}
      \begin{itemize}
        \item Development of measurement equipment for characterizing\\ radio channels from 0 Hz to 300 GHz
        \item Measurement, simulation, and modeling of radio channels
        \item Analysis of radio channel characteristics using signal\\ processing - high-resolution estimation methods and AI
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.5cm}
  \begin{itemize}
    \item Exploitation of radio channel characteristics for environmental sensing - radar technology\\ and \ac{acro:icas}
    \item Emulation of electromagnetic wave propagation for testing mobile and satellite communications
    \item Standardization and development of solutions relevant to mobile communications standards
  \end{itemize}
\end{frame}
